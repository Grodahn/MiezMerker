add_library(firmware-core
  src/core.cpp
  src/node_identity.cpp
  src/memory_stores.cpp
  src/protocol_view.cpp
  src/types.cpp
  src/ble_codec.cpp
  src/ble_sync.cpp
)
target_include_directories(firmware-core PUBLIC include)
if(MSVC)
  target_compile_options(firmware-core PRIVATE /W4 /WX)
else()
  target_compile_options(firmware-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(core-test tests/core_test.cpp)
  target_link_libraries(core-test PRIVATE firmware-core)
  add_test(NAME firmware-core COMMAND core-test)
  set_tests_properties(firmware-core PROPERTIES LABELS "lifecycle;storage")
  add_executable(node-identity-test tests/node_identity_test.cpp)
  target_link_libraries(node-identity-test PRIVATE firmware-core)
  add_test(NAME node-identity COMMAND node-identity-test)
  set_tests_properties(node-identity PROPERTIES LABELS "lifecycle;storage")

  add_executable(observation-test tests/observation_test.cpp)
  target_link_libraries(observation-test PRIVATE firmware-core)
  if(MSVC)
    target_compile_options(observation-test PRIVATE /W4 /WX)
  else()
    target_compile_options(observation-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME firmware-observations COMMAND observation-test)
  set_tests_properties(firmware-observations PROPERTIES LABELS "storage")
  add_executable(recovery-test tests/recovery_test.cpp)
  target_link_libraries(recovery-test PRIVATE firmware-core)
  if(MSVC)
    target_compile_options(recovery-test PRIVATE /W4 /WX)
  else()
    target_compile_options(recovery-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME firmware-recovery COMMAND recovery-test)
  set_tests_properties(firmware-recovery PROPERTIES LABELS "storage;lifecycle")
  add_executable(ble-sync-test tests/ble_sync_test.cpp)
  target_link_libraries(ble-sync-test PRIVATE firmware-core)
  if(MSVC)
    target_compile_options(ble-sync-test PRIVATE /W4 /WX)
  else()
    target_compile_options(ble-sync-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME firmware-ble-sync COMMAND ble-sync-test)
  set_tests_properties(firmware-ble-sync PROPERTIES LABELS "ble;storage")
  # Assertions include operations under test; keep them in Release/CI builds.
  foreach(test_target core-test node-identity-test observation-test recovery-test ble-sync-test)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
